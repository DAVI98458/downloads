local Config = {
    Enabled = false,
    BallNames = {"MPS", "VEF", "TPS", "VCL", "VRF", "UEFA", "VCA", "VRF", "CBM", "SAML", "OFA", "VCL", "VFA", "AIFA"},
    Prediction = 0.16,
    Radius = 1.6,
    LerpWeight = 0.15,
    MaxVelocity = 120,
    DeadzoneVel = 1.5,
    SearchInterval = 0.1,
    BallOverlapThreshold = 1.3,
    RecoveryLerp = 0.55,
    RecoveryFrames = 10,

    -- [[ SHIELDING CONFIG ]] --
    ShieldingBias = 50,          
    ShieldActivationDist = 9.0,   

    -- [[ AUTO MOSS CONFIG ]] --
    MossEnabled = false,
    MossRadius = 200,
    HeaderHeight = 5.5,
    MossPredictionTime = 2.5
}

local CurrentBall = nil
local lastPos = Vector3.new(0, 0, 0)
local initialized = false
local recoveryCounter = 0

local function SafeDir(v, fallback)
    local flat = Vector3.new(v.X, 0, v.Z)
    if flat.Magnitude > 0.01 then return flat.Unit end
    local fb = Vector3.new(fallback.X, 0, fallback.Z)
    if fb.Magnitude > 0.01 then return fb.Unit end
    return Vector3.new(1, 0, 0)
end

local function GetClosestOpponent(ballPos)
    local closestEnemy = nil
    local minDist = math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Team ~= LocalPlayer.Team then
            local char = player.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root then
                local dist = (root.Position - ballPos).Magnitude
                if dist < minDist then
                    minDist = dist
                    closestEnemy = root
                end
            end
        end
    end
    return closestEnemy, minDist
end

local function GetShieldTarget(root, ballPos)
    local enemyRoot, enemyDist = GetClosestOpponent(ballPos)
    local defaultDir = SafeDir(root.Position - ballPos, root.CFrame.LookVector)
    
    if not enemyRoot or enemyDist > Config.ShieldActivationDist or Config.ShieldingBias == 0 then
        return ballPos + (defaultDir * Config.Radius)
    end
    
    local shieldDir = SafeDir(ballPos - enemyRoot.Position, root.CFrame.LookVector)
    local biasAlpha = math.clamp(Config.ShieldingBias / 100, 0, 1)
    local finalDir = defaultDir:Lerp(shieldDir, biasAlpha).Unit
    return ballPos + (finalDir * Config.Radius)
end

local function GetRecoveryTarget(root, ballPos)
    local awayBall = SafeDir(root.Position - ballPos, root.CFrame.LookVector)
    return ballPos + (awayBall * Config.Radius)
end

-- [[ AUTO MOSS PREDICTION LOGIC ]] --
local function GetMossTarget(root, ball)
    local p0 = ball.Position
    local v0 = ball.Velocity
    local g = workspace.Gravity
    
    local targetY = root.Position.Y + Config.HeaderHeight
    local a = -0.5 * g
    local b = v0.Y
    local c = p0.Y - targetY
    
    local discriminant = (b * b) - (4 * a * c)
    local t = Config.MossPredictionTime
    
    if discriminant >= 0 and math.abs(a) > 0.001 then
        local t1 = (-b + math.sqrt(discriminant)) / (2 * a)
        local t2 = (-b - math.sqrt(discriminant)) / (2 * a)
        
        local validTimes = {}
        if t1 > 0 then table.insert(validTimes, t1) end
        if t2 > 0 then table.insert(validTimes, t2) end
        
        if #validTimes > 0 then
            table.sort(validTimes)
            t = validTimes[1] -- Intercept at earliest positive time frame
        end
    end
    
    t = math.clamp(t, 0, Config.MossPredictionTime)
    local predictedPos = p0 + (v0 * t) + (0.5 * Vector3.new(0, -g, 0) * (t * t))
    return Vector3.new(predictedPos.X, root.Position.Y, predictedPos.Z)
end

task.spawn(function()
    while task.wait(Config.SearchInterval) do
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then continue end
        local best, minDist = nil, math.huge

        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                for _, name in ipairs(Config.BallNames) do
                    if obj.Name == name then
                        local d = (obj.Position - root.Position).Magnitude
                        if d < minDist then
                            minDist = d
                            best = obj
                        end
                        break
                    end
                end
            end
        end

        CurrentBall = best
        if best and not initialized then
            lastPos = best.Position + SafeDir(
                (char.HumanoidRootPart and char.HumanoidRootPart.Position or Vector3.new(1,0,0)) - best.Position,
                Vector3.new(1, 0, 0)
            ) * Config.Radius
            initialized = true
        end
    end
end)

RunService.Heartbeat:Connect(function(dt)
    if not Config.Enabled and not Config.MossEnabled then return end
    local Char = LocalPlayer.Character
    local Root = Char and Char:FindFirstChild("HumanoidRootPart")
    local Hum  = Char and Char:FindFirstChild("Humanoid")
    if not Root or not Hum or not CurrentBall or not CurrentBall.Parent then return end
    
    local rawVel = CurrentBall.Velocity
    local velMag = rawVel.Magnitude
    local clampedVel = velMag > Config.MaxVelocity and (rawVel.Unit * Config.MaxVelocity) or rawVel
    local ballPos = velMag > Config.DeadzoneVel and CurrentBall.Position + (clampedVel * Config.Prediction) or CurrentBall.Position
    local flatDist = Vector3.new(Root.Position.X - CurrentBall.Position.X, 0, Root.Position.Z - CurrentBall.Position.Z).Magnitude
    
    local targetPos, activeLerp
    local mossingActive = false
    
    if Config.Enabled and flatDist < Config.BallOverlapThreshold then 
        recoveryCounter = Config.RecoveryFrames 
    end
    
    if Config.Enabled and recoveryCounter > 0 then
        recoveryCounter -= 1
        targetPos = GetRecoveryTarget(Root, CurrentBall.Position)
        activeLerp = Config.RecoveryLerp
    elseif Config.MossEnabled and (CurrentBall.Position - Root.Position).Magnitude <= Config.MossRadius and CurrentBall.Position.Y > (Root.Position.Y + 2.5) then
        -- Execute Auto Moss positioning via MoveTo when active
        targetPos = GetMossTarget(Root, CurrentBall)
        activeLerp = 0.35
        mossingActive = true
    elseif Config.Enabled then
        targetPos = GetShieldTarget(Root, ballPos)
        local d = (targetPos - Root.Position).Magnitude
        activeLerp = math.clamp(Config.LerpWeight * (d / 8), 0.06, 0.5)
    end
    
    if targetPos then
        lastPos = lastPos:Lerp(targetPos, activeLerp)
        
        if not mossingActive and Config.Enabled then
            local toBall = Vector3.new(lastPos.X - CurrentBall.Position.X, 0, lastPos.Z - CurrentBall.Position.Z)
            if toBall.Magnitude < Config.Radius then
                local pushDir = toBall.Magnitude > 0.01 and toBall.Unit or SafeDir(Root.Position - CurrentBall.Position, Root.CFrame.LookVector)
                lastPos = Vector3.new(CurrentBall.Position.X + pushDir.X * Config.Radius, lastPos.Y, CurrentBall.Position.Z + pushDir.Z * Config.Radius)
            end
        end
        
        local goal = Vector3.new(lastPos.X, Root.Position.Y, lastPos.Z)
        if (goal - Root.Position).Magnitude > 0.1 then 
            Hum:MoveTo(goal) 
        end
    end
end)

        if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
            dragInput = i
        end
        if dragging and i == dragInput then
            local d = Vector2.new(i.Position.X, i.Position.Y) - dragStart
            Panel.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)

    UpdateUI()
end

LocalPlayer:WaitForChild("PlayerGui")
