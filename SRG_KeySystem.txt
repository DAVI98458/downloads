local function loadSavedKey()
    local ok, val = pcall(function() return readfile(KEY_FILE) end)

-- ══════════════════════════════════════════
--  HWID HASH — matches website JS exactly
--  h = (h * 31 + charcode) % (36^6)
--  result as 6-char uppercase base36
-- ══════════════════════════════════════════
local B36    = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
local SECRET = "SRG_8x2k$vNp"

local function toBase36(n)
    n = math.floor(n or 0)
    if n == 0 then return "0" end
    local s = ""
    while n > 0 do
        local rem = n % 36
        s = B36:sub(rem + 1, rem + 1) .. s
        n = math.floor(n / 36)
    end
    return s
end

local function fromBase36(s)
    local n = 0
    for i = 1, #s do
        local c = s:sub(i, i):upper()
        local v = B36:find(c, 1, true)
        if not v then return nil end
        n = n * 36 + (v - 1)
    end
    return n
end

local function deriveTag(dayStr)
    dayStr = tostring(dayStr or "")
    local raw = dayStr .. SECRET
    local h   = 0
    for i = 1, #raw do
        h = ((h * 31) + raw:byte(i)) % 4294967296
    end
    local result = toBase36(math.floor(h) % 1679616) or "0000"
    while #result < 4 do result = "0" .. result end
    return result:upper()
end

local FIREBASE_URL = "https://scripts-ca84b-default-rtdb.firebaseio.com"  -- sem barra no final
local HttpService  = game:GetService("HttpService")

local function fbKeyPath(key)
    return key:gsub("[%.%#%$%[%]%/]", "_")
end

local function fbGet(key)
    local ok, res = pcall(function()
        return HttpService:GetAsync(FIREBASE_URL .. "/keys/" .. fbKeyPath(key) .. ".json", true)
    end)
    if not ok or not res or res == "null" then return nil end
    local dok, data = pcall(function() return HttpService:JSONDecode(res) end)
    if not dok then return nil end
    return data
end

local function fbPatch(key, data)
    pcall(function()
        HttpService:RequestAsync({
            Url     = FIREBASE_URL .. "/keys/" .. fbKeyPath(key) .. ".json",
            Method  = "PATCH",
            Headers = { ["Content-Type"] = "application/json" },
            Body    = HttpService:JSONEncode(data)
        })
    end)
end

-- UTC day (força UTC independente do timezone do sistema)
local function getUTCDay()
    local t = os.date("!*t")
    local utcTs = os.time({
        year=t.year, month=t.month, day=t.day,
        hour=t.hour, min=t.min, sec=t.sec, isdst=false
    })
    return math.floor(utcTs / 86400)
end

local function getMyUserId()
    local ok, id = pcall(function()
        return tostring(game:GetService("Players").LocalPlayer.UserId)
    end)
    return ok and id or "0"
end

-- helper: check if value is null (nil or JSONNull)
local function isNull(v)
    if v == nil then return true end
    local ok, res = pcall(function() return v == HttpService.JSONNull end)
    return ok and res
end

-- patch and verify with re-read (Bug 2 fix)
local function fbPatchVerified(key, data)
    fbPatch(key, data)
    task.wait(0.6)
    local verify = fbGet(key)
    return verify ~= nil and not isNull(verify.userId)
end

-- ══════════════════════════════════════════
--  CONNECTIVITY CHECK
-- ══════════════════════════════════════════
local function checkConnectivity()
    local ok, res = pcall(function()
        return HttpService:GetAsync(FIREBASE_URL .. "/.json?shallow=true", true)
    end)
    if not ok then
        return false, "HTTP requests may be disabled in your executor, or firebaseio.com is blocked."
    end
    if res == nil or res == "" then
        return false, "Could not reach the key server."
    end
    return true, "ok"
end

    local dayPart, second = key:match("^SRG%-([A-Z0-9]+)%-([A-Z0-9]+)$")
